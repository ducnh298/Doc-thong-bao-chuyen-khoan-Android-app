# ADB Test Commands - Giả lập thông báo chuyển khoản
#
# Format đúng:
#   adb shell 'cmd notification post -S bigtext -t "<title>" <tag> "<text>"'
#
# Lý do:
#   - Bọc single quote '' → Android shell giữ nguyên double quote bên trong
#   - Dùng -t (không phải --title hay --text vì --text không phải flag hợp lệ)
#   - <tag> là tên định danh tuỳ ý, <text> là nội dung thông báo (positional arg)
#
# Parser yêu cầu:
#   -t (title) phải chứa VND và dấu +/-
#   <text>     phải chứa SD

# ── Tiền đến ─────────────────────────────────────────────────────────────────
adb shell 'cmd notification post -S bigtext -t "MB Bank +500,000 VND" mb_in "TK:1234567890 +500,000VND SD:2,000,000VND"'

adb shell 'cmd notification post -S bigtext -t "Vietcombank +1,200,000 VND" vcb_in "TK:0011234567 +1,200,000VND SD:5,000,000VND"'

adb shell 'cmd notification post -S bigtext -t "Techcombank +350,000 VND" tcb_in "TK:9876543210 +350,000VND SD:3,000,000VND"'

adb shell 'cmd notification post -S bigtext -t "Agribank +2,500,000 VND" agri_in "TK:12345678901 +2,500,000VND SD:10,000,000VND"'

adb shell 'cmd notification post -S bigtext -t "BIDV +800,000 VND" bidv_in "TK:31410001234567 +800,000VND SD:4,500,000VND"'

# ── Tiền đi ──────────────────────────────────────────────────────────────────
adb shell 'cmd notification post -S bigtext -t "MB Bank -200,000 VND" mb_out "TK:1234567890 -200,000VND SD:1,800,000VND"'

adb shell 'cmd notification post -S bigtext -t "Vietcombank -500,000 VND" vcb_out "TK:0011234567 -500,000VND SD:4,500,000VND"'

# ── Số tiền lớn ───────────────────────────────────────────────────────────────
adb shell 'cmd notification post -S bigtext -t "MB Bank +10,500,000 VND" mb_big "TK:1234567890 +10,500,000VND SD:50,000,000VND"'

# ── Xoá noti ─────────────────────────────────────────────────────────────────
adb shell cmd notification cancel mb_in
adb shell cmd notification cancel mb_out

# ── Debug log ─────────────────────────────────────────────────────────────────
adb logcat -s MainActivity
